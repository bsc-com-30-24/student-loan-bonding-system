import uuid

from django.conf import settings
from django.db import models

from bonds.models import Bond


class Agreement(models.Model):

    class Status(models.TextChoices):
        GENERATED = 'GENERATED', 'Generated'
        PENDING_SIGNATURE = 'PENDING_SIGNATURE', 'Pending Signature'
        SIGNED = 'SIGNED', 'Signed'
        COUNTERSIGNED = 'COUNTERSIGNED', 'Countersigned'

    agreement_number = models.UUIDField(
        primary_key=True,
        default=uuid.uuid4,
        editable=False
    )

    # Each bond can have only one agreement.
    # Nullable initially to preserve existing agreement records.
    bond = models.OneToOneField(
        Bond,
        on_delete=models.PROTECT,
        related_name='agreement',
        null=True,
        blank=True
    )

    student_id = models.CharField(max_length=50)

    loan_reference = models.CharField(max_length=100)

    agreement_content = models.TextField()

    # The unsigned PDF generated by the system.
    generated_pdf = models.FileField(
        upload_to='agreements/pdfs/',
        null=True,
        blank=True
    )

    status = models.CharField(
        max_length=30,
        choices=Status.choices,
        default=Status.GENERATED
    )

    generated_at = models.DateTimeField(auto_now_add=True)

    # The student downloads the generated PDF, signs it and uploads it here.
    student_signed_document = models.FileField(
        upload_to='agreements/student_signed/',
        null=True,
        blank=True
    )

    # SHA-256 of the uploaded file, kept as tamper evidence.
    student_signed_document_sha256 = models.CharField(
        max_length=64,
        blank=True,
        default=''
    )

    student_signed_at = models.DateTimeField(
        null=True,
        blank=True
    )

    # The Loan Board Officer downloads the student-signed PDF, signs it
    # and uploads the final copy here.
    countersigned_document = models.FileField(
        upload_to='agreements/countersigned/',
        null=True,
        blank=True
    )

    countersigned_document_sha256 = models.CharField(
        max_length=64,
        blank=True,
        default=''
    )

    countersigned_by = models.ForeignKey(
        settings.AUTH_USER_MODEL,
        on_delete=models.SET_NULL,
        null=True,
        blank=True,
        related_name='countersigned_agreements'
    )

    board_countersigned_at = models.DateTimeField(
        null=True,
        blank=True
    )

    created_at = models.DateTimeField(auto_now_add=True)

    updated_at = models.DateTimeField(auto_now=True)

    def __str__(self):
        return str(self.agreement_number)

from io import BytesIO

from reportlab.lib.pagesizes import A4
from reportlab.lib.styles import getSampleStyleSheet
from reportlab.lib.units import mm
from reportlab.platypus import (
    SimpleDocTemplate,
    Paragraph,
    Spacer,
    Preformatted,
)


def generate_agreement_pdf(agreement):

    buffer = BytesIO()

    document = SimpleDocTemplate(
        buffer,
        pagesize=A4,
        rightMargin=20 * mm,
        leftMargin=20 * mm,
        topMargin=20 * mm,
        bottomMargin=20 * mm,
    )

    styles = getSampleStyleSheet()

    story = []

    story.append(
        Paragraph(
            "STUDENT LOAN BONDING AGREEMENT",
            styles["Title"],
        )
    )

    story.append(Spacer(1, 15))

    story.append(
        Paragraph(
            f"Agreement Number: {agreement.agreement_number}",
            styles["Normal"],
        )
    )

    story.append(
        Paragraph(
            f"Bond Reference: {agreement.bond.bond_reference}",
            styles["Normal"],
        )
    )

    story.append(Spacer(1, 20))

    story.append(
        Preformatted(
            agreement.agreement_content,
            styles["Code"],
        )
    )

    story.append(Spacer(1, 30))

    story.append(
        Paragraph(
            "Student Signature: __________________________",
            styles["Normal"],
        )
    )

    story.append(Spacer(1, 20))

    story.append(
        Paragraph(
            "Loan Board Officer Signature: __________________________",
            styles["Normal"],
        )
    )

    document.build(story)

    pdf_data = buffer.getvalue()

    buffer.close()

    return pdf_data

import shutil
import tempfile

from django.contrib.auth import get_user_model
from django.core.files.uploadedfile import SimpleUploadedFile
from django.test import override_settings
from django.urls import reverse
from rest_framework.test import APITestCase

from agreements.models import Agreement
from authentication.models import UserProfile
from bonds.models import Bond

from .models import Document, Notification
from .services import COMPULSORY_DOCUMENT_TYPES

User = get_user_model()

TEST_MEDIA_ROOT = tempfile.mkdtemp()


def make_user(username, role, registration_number=None):
    """
    ASSUMPTION: UserProfile has a `user` link, a `role` field and a
    `student_registration_number` field. If your model differs, this is the
    only place to adjust.
    """
    user = User.objects.create_user(username=username, password="Pass12345!")
    profile, _ = UserProfile.objects.get_or_create(user=user)
    profile.role = role
    if registration_number:
        profile.student_registration_number = registration_number
    profile.save()
    return user


def signed_pdf(name="signed.pdf"):
    return SimpleUploadedFile(
        name, b"%PDF-1.4 signed copy", content_type="application/pdf"
    )


@override_settings(MEDIA_ROOT=TEST_MEDIA_ROOT)
class WorkflowIntegrationTests(APITestCase):

    @classmethod
    def tearDownClass(cls):
        super().tearDownClass()
        shutil.rmtree(TEST_MEDIA_ROOT, ignore_errors=True)

    def setUp(self):
        self.student = make_user(
            "student1", UserProfile.Role.STUDENT, "REG-001"
        )
        self.other_student = make_user(
            "student2", UserProfile.Role.STUDENT, "REG-002"
        )
        self.admin = make_user("admin1", UserProfile.Role.UNIVERSITY_ADMIN)
        self.officer = make_user(
            "officer1", UserProfile.Role.LOAN_BOARD_OFFICER
        )

    # ---------- helpers ----------

    def upload(self, doc_type, user=None):
        self.client.force_authenticate(user or self.student)
        response = self.client.post(
            reverse("student_documents_api"),
            {
                "document_type": doc_type,
                "file": SimpleUploadedFile(
                    f"{doc_type}.pdf",
                    b"%PDF-1.4 test file",
                    content_type="application/pdf",
                ),
            },
            format="multipart",
        )
        self.assertEqual(response.status_code, 201, response.data)
        return response.data["id"]

    def review(self, doc_id, new_status="verified", reason=""):
        self.client.force_authenticate(self.admin)
        return self.client.patch(
            reverse("document_verification_api", kwargs={"pk": doc_id}),
            {"status": new_status, "rejection_reason": reason},
            format="json",
        )

    def upload_and_verify_all(self, user=None):
        """Returns the response of the LAST verification."""
        response = None
        for doc_type in COMPULSORY_DOCUMENT_TYPES:
            doc_id = self.upload(doc_type, user)
            response = self.review(doc_id)
            self.assertEqual(response.status_code, 200, response.data)
        return response

    def approve_bond(self, bond_id):
        self.client.force_authenticate(self.admin)
        return self.client.patch(
            reverse("bond-status-update", kwargs={"pk": bond_id}),
            {"status": "approved"},
            format="json",
        )

    # ---------- tests ----------

    def test_full_workflow(self):
        # 1-2. Student submits, admin verifies. The bond appears only
        # after the last compulsory document is verified.
        last = None
        for index, doc_type in enumerate(COMPULSORY_DOCUMENT_TYPES):
            last = self.review(self.upload(doc_type))
            self.assertEqual(last.status_code, 200, last.data)
            is_last = index == len(COMPULSORY_DOCUMENT_TYPES) - 1
            self.assertEqual(last.data["bond_generated"], is_last)
            self.assertEqual(Bond.objects.count(), 1 if is_last else 0)

        # 3. Exactly one pending bond, reported in the response.
        bond = Bond.objects.get()
        self.assertEqual(bond.student, self.student)
        self.assertEqual(bond.status, "pending")
        self.assertEqual(last.data["bond_id"], bond.id)

        # Notifications: one per review plus one for the bond.
        self.assertEqual(
            Notification.objects.filter(student=self.student).count(),
            len(COMPULSORY_DOCUMENT_TYPES) + 1,
        )

        # The student sees the bond; another student does not.
        self.client.force_authenticate(self.student)
        listing = self.client.get(reverse("bond-list-create"))
        self.assertEqual([b["id"] for b in listing.data], [bond.id])
        self.client.force_authenticate(self.other_student)
        self.assertEqual(
            self.client.get(reverse("bond-list-create")).data, []
        )

        # 5. No agreement before the bond is approved.
        self.client.force_authenticate(self.student)
        response = self.client.post(reverse("agreement-generate"))
        self.assertEqual(response.status_code, 400)

        # 4. Admin approves the bond.
        response = self.approve_bond(bond.id)
        self.assertEqual(response.status_code, 200, response.data)

        # 5. Student generates the agreement (once only).
        self.client.force_authenticate(self.student)
        response = self.client.post(reverse("agreement-generate"))
        self.assertEqual(response.status_code, 201, response.data)
        agreement_number = response.data["agreement_number"]
        self.assertEqual(Agreement.objects.get().bond_id, bond.id)
        self.assertEqual(
            self.client.post(reverse("agreement-generate")).status_code, 409
        )

        sign_url = reverse(
            "agreement-sign", kwargs={"agreement_number": agreement_number}
        )
        countersign_url = reverse(
            "agreement-countersign",
            kwargs={"agreement_number": agreement_number},
        )

        download_url = reverse(
            "agreement-download",
            kwargs={"agreement_number": agreement_number},
        )

        def upload(url, user, file=None):
            self.client.force_authenticate(user)
            data = {"signed_document": file} if file else {}
            return self.client.post(url, data, format="multipart")

        # Countersigning before the student signs is refused.
        self.assertEqual(
            upload(countersign_url, self.officer, signed_pdf()).status_code,
            400,
        )

        # Another student cannot sign it.
        self.assertEqual(
            upload(sign_url, self.other_student, signed_pdf()).status_code,
            403,
        )

        # The student downloads the original; no signed copy exists yet.
        self.client.force_authenticate(self.student)
        self.assertEqual(self.client.get(download_url).status_code, 200)
        self.assertEqual(
            self.client.get(
                download_url, {"version": "student_signed"}
            ).status_code,
            404,
        )
        self.assertEqual(
            self.client.get(download_url, {"version": "bogus"}).status_code,
            400,
        )

        # A missing file or a non-PDF upload is refused.
        self.assertEqual(upload(sign_url, self.student).status_code, 400)
        not_pdf = SimpleUploadedFile(
            "signed.txt", b"not a pdf", content_type="text/plain"
        )
        self.assertEqual(
            upload(sign_url, self.student, not_pdf).status_code, 400
        )
        fake_pdf = SimpleUploadedFile(
            "signed.pdf", b"this is not really a pdf",
            content_type="application/pdf",
        )
        self.assertEqual(
            upload(sign_url, self.student, fake_pdf).status_code, 400
        )

        # 6. The owner uploads the signed copy.
        response = upload(sign_url, self.student, signed_pdf())
        self.assertEqual(response.status_code, 200, response.data)
        agreement = Agreement.objects.get()
        self.assertEqual(agreement.status, Agreement.Status.SIGNED)
        self.assertTrue(agreement.student_signed_document)
        self.assertEqual(len(agreement.student_signed_document_sha256), 64)

        # A second upload is refused.
        self.assertEqual(
            upload(sign_url, self.student, signed_pdf()).status_code, 400
        )

        # Only the Loan Board Officer may countersign.
        for user in (self.student, self.admin):
            self.assertEqual(
                upload(countersign_url, user, signed_pdf()).status_code, 403
            )

        # The officer downloads the student-signed copy, signs it and
        # uploads the final version.
        self.client.force_authenticate(self.officer)
        self.assertEqual(
            self.client.get(
                download_url, {"version": "student_signed"}
            ).status_code,
            200,
        )
        self.assertEqual(
            upload(countersign_url, self.officer).status_code, 400
        )

        # 7. The officer uploads the countersigned copy.
        response = upload(countersign_url, self.officer, signed_pdf())
        self.assertEqual(response.status_code, 200, response.data)
        agreement.refresh_from_db()
        self.assertEqual(agreement.status, Agreement.Status.COUNTERSIGNED)
        self.assertTrue(agreement.countersigned_document)
        self.assertEqual(agreement.countersigned_by, self.officer)

        # The final copy can be downloaded by everyone involved, but not
        # by an unrelated student.
        for user in (self.student, self.officer, self.admin):
            self.client.force_authenticate(user)
            self.assertEqual(
                self.client.get(
                    download_url, {"version": "countersigned"}
                ).status_code,
                200,
            )
        self.client.force_authenticate(self.other_student)
        self.assertEqual(
            self.client.get(
                download_url, {"version": "countersigned"}
            ).status_code,
            403,
        )

    def test_rejected_document_blocks_the_bond(self):
        for doc_type in COMPULSORY_DOCUMENT_TYPES[:-1]:
            self.review(self.upload(doc_type))

        rejected = self.review(
            self.upload(COMPULSORY_DOCUMENT_TYPES[-1]),
            "rejected",
            "Photo is blurry",
        )
        self.assertEqual(rejected.status_code, 200)
        self.assertEqual(Bond.objects.count(), 0)

        # Re-uploading and verifying a good copy completes the set.
        response = self.review(self.upload(COMPULSORY_DOCUMENT_TYPES[-1]))
        self.assertTrue(response.data["bond_generated"])
        self.assertEqual(Bond.objects.count(), 1)

    def test_no_duplicate_bond(self):
        self.upload_and_verify_all()
        self.assertEqual(Bond.objects.count(), 1)

        # Re-verifying more copies of compulsory documents adds nothing.
        for doc_type in COMPULSORY_DOCUMENT_TYPES:
            response = self.review(self.upload(doc_type))
            self.assertFalse(response.data["bond_generated"])

        self.assertEqual(Bond.objects.filter(student=self.student).count(), 1)

    def test_bonds_are_per_student(self):
        self.upload_and_verify_all(self.student)
        self.upload_and_verify_all(self.other_student)
        self.assertEqual(Bond.objects.count(), 2)

    def test_roles_are_enforced_on_documents(self):
        doc_id = self.upload(COMPULSORY_DOCUMENT_TYPES[0])
        verify_url = reverse("document_verification_api", kwargs={"pk": doc_id})

        for user in (self.student, self.officer):
            self.client.force_authenticate(user)
            response = self.client.patch(
                verify_url, {"status": "verified"}, format="json"
            )
            self.assertEqual(response.status_code, 403)

        for user in (self.admin, self.officer):
            self.client.force_authenticate(user)
            response = self.client.get(reverse("student_documents_api"))
            self.assertEqual(response.status_code, 403)

        self.client.force_authenticate(None)
        response = self.client.get(reverse("student_documents_api"))
        self.assertEqual(response.status_code, 401)

    def test_bond_review_is_admin_only(self):
        self.upload_and_verify_all()
        bond = Bond.objects.get()

        for user in (self.student, self.officer):
            self.client.force_authenticate(user)
            response = self.client.patch(
                reverse("bond-status-update", kwargs={"pk": bond.id}),
                {"status": "approved"},
                format="json",
            )
            self.assertEqual(response.status_code, 403)

import os

from django.core.exceptions import ValidationError

MAX_SIGNED_UPLOAD_SIZE = 10 * 1024 * 1024  # 10 MB


def validate_signed_pdf(uploaded_file):
    """A signed agreement must be a real, non-empty PDF up to 10 MB."""
    extension = os.path.splitext(uploaded_file.name)[1].lower()

    if extension != ".pdf":
        raise ValidationError("Upload the signed agreement as a PDF file.")

    if uploaded_file.size == 0:
        raise ValidationError("The uploaded file is empty.")

    if uploaded_file.size > MAX_SIGNED_UPLOAD_SIZE:
        raise ValidationError("File is too large. The maximum size is 10 MB.")

    header = uploaded_file.read(5)
    uploaded_file.seek(0)

    if header != b"%PDF-":
        raise ValidationError("The uploaded file is not a valid PDF.")

def get_registration_number(user):
    """Return the student's registration number, or None if not set."""
    profile = getattr(user, "profile", None)
    return getattr(profile, "student_registration_number", None) or None


def get_student_identifiers(user):
    """All values an Agreement.student_id may hold for this user."""
    identifiers = [str(user.pk), user.get_username()]
    registration_number = get_registration_number(user)
    if registration_number:
        identifiers.append(str(registration_number))
    return identifiers


def agreements_for_user(user):
    """Students see their own agreements; admins and officers see all."""
    role = get_user_role(user)

    if role == UserProfile.Role.STUDENT:
        return Agreement.objects.filter(
            student_id__in=get_student_identifiers(user)
        ).order_by("-generated_at")

    if role in (
        UserProfile.Role.UNIVERSITY_ADMIN,
        UserProfile.Role.LOAN_BOARD_OFFICER,
    ):
        return Agreement.objects.all().order_by("-generated_at")

    return Agreement.objects.none()


def sha256_of(uploaded_file):
    """SHA-256 hex digest of an uploaded file (leaves it rewound)."""
    digest = hashlib.sha256()
    for chunk in uploaded_file.chunks():
        digest.update(chunk)
    uploaded_file.seek(0)
    return digest.hexdigest()

class AgreementGenerateView(APIView):
    permission_classes = [permissions.IsAuthenticated, IsStudent]

    @extend_schema(
        request=None,
        responses={
            201: AgreementSerializer,
            400: ErrorResponseSerializer,
            403: ErrorResponseSerializer,
            409: ErrorResponseSerializer,
        },
        summary="Generate agreement from an approved bond",
        description=(
            "Generates an agreement and its PDF for the authenticated "
            "student. The student must have an approved bond. The system "
            "obtains the bond automatically; no student ID or bond "
            "reference is accepted from the request."
        ),
        tags=["Agreements"],
    )
    def post(self, request):
        with transaction.atomic():
            bond = (
                Bond.objects.select_for_update()
                .filter(student=request.user, status="approved")
                .first()
            )

            if bond is None:
                return Response(
                    {
                        "error": (
                            "You cannot generate an agreement because "
                            "you do not have an approved bond."
                        )
                    },
                    status=status.HTTP_400_BAD_REQUEST,
                )

            # Prevent duplicate agreements for the same bond.
            if Agreement.objects.filter(bond=bond).exists():
                return Response(
                    {
                        "error": (
                            "An agreement has already been generated "
                            "for this bond."
                        )
                    },
                    status=status.HTTP_409_CONFLICT,
                )

            registration_number = (
                get_registration_number(request.user)
                or str(request.user.pk)
            )

            # The Bond model has bond_reference but no separate
            # loan_reference, so the bond reference is used for now.
            loan_reference = str(bond.bond_reference)

            agreement_content = (
                "STUDENT LOAN BONDING AGREEMENT\n\n"
                f"Student ID: {registration_number}\n"
                f"Bond Reference: {bond.bond_reference}\n\n"
                "This agreement confirms the student's bonding "
                "arrangement following approval of the bond.\n\n"
                "The final approved agreement terms must be inserted "
                "before the agreement is used officially."
            )

            agreement = Agreement.objects.create(
                bond=bond,
                student_id=str(registration_number),
                loan_reference=loan_reference,
                agreement_content=agreement_content,
                status=Agreement.Status.GENERATED,
            )

            # Generate and store the PDF. Inside the transaction so a
            # failure here rolls back the agreement as well.
            pdf_data = generate_agreement_pdf(agreement)
            agreement.generated_pdf.save(
                f"agreement_{agreement.agreement_number}.pdf",
                ContentFile(pdf_data),
                save=True,
            )

        return Response(
            AgreementSerializer(agreement).data,
            status=status.HTTP_201_CREATED,
        )

class AgreementSignView(APIView):
    permission_classes = [permissions.IsAuthenticated, IsStudent]
    parser_classes = [parsers.MultiPartParser, parsers.FormParser]

    @extend_schema(
        request=AgreementSignedUploadSerializer,
        parameters=[AGREEMENT_NUMBER_PARAM],
        responses={
            200: AgreementActionResponseSerializer,
            400: ErrorResponseSerializer,
            403: ErrorResponseSerializer,
            404: ErrorResponseSerializer,
        },
        summary="Student uploads the signed agreement",
        description=(
            "Step 1 of signing: download the agreement with "
            "GET /download/ (version 'original'), sign it, then upload the "
            "signed PDF here as multipart/form-data in the 'signed_document' "
            "field. Only the student who owns the agreement can do this, "
            "once, while it is GENERATED or PENDING_SIGNATURE. The agreement "
            "then becomes SIGNED and waits for the Loan Board Officer."
        ),
        tags=["Agreements"],
    )
    def post(self, request, agreement_number):
        with transaction.atomic():
            agreement = get_object_or_404(
                Agreement.objects.select_for_update(),
                agreement_number=agreement_number,
            )

            if agreement.student_id not in get_student_identifiers(
                request.user
            ):
                return Response(
                    {
                        "error": (
                            "You are not authorized to sign "
                            "this agreement."
                        )
                    },
                    status=status.HTTP_403_FORBIDDEN,
                )

            if agreement.status not in (
                Agreement.Status.GENERATED,
                Agreement.Status.PENDING_SIGNATURE,
            ):
                return Response(
                    {
                        "error": (
                            "This agreement cannot be signed "
                            "in its current status."
                        )
                    },
                    status=status.HTTP_400_BAD_REQUEST,
                )

            serializer = AgreementSignedUploadSerializer(data=request.data)
            serializer.is_valid(raise_exception=True)
            uploaded = serializer.validated_data["signed_document"]

            agreement.student_signed_document_sha256 = sha256_of(uploaded)
            agreement.student_signed_document.save(
                f"agreement_{agreement.agreement_number}_student_signed.pdf",
                uploaded,
                save=False,
            )
            agreement.status = Agreement.Status.SIGNED
            agreement.student_signed_at = timezone.now()
            agreement.save(
                update_fields=[
                    "student_signed_document",
                    "student_signed_document_sha256",
                    "status",
                    "student_signed_at",
                    "updated_at",
                ]
            )

        return Response(
            {
                "message": "Signed agreement uploaded successfully.",
                "agreement": AgreementSerializer(agreement).data,
            },
            status=status.HTTP_200_OK,
        )

class AgreementCountersignView(APIView):
    permission_classes = [permissions.IsAuthenticated, IsLoanBoardOfficer]
    parser_classes = [parsers.MultiPartParser, parsers.FormParser]

    @extend_schema(
        request=AgreementSignedUploadSerializer,
        parameters=[AGREEMENT_NUMBER_PARAM],
        responses={
            200: AgreementActionResponseSerializer,
            400: ErrorResponseSerializer,
            403: ErrorResponseSerializer,
            404: ErrorResponseSerializer,
        },
        summary="Loan Board Officer uploads the countersigned agreement",
        description=(
            "Step 2 of signing: download the student-signed copy with "
            "GET /download/?version=student_signed, countersign it, then "
            "upload the final PDF here as multipart/form-data in the "
            "'signed_document' field. Only a Loan Board Officer can do "
            "this, and only after the student has uploaded their signed "
            "copy. The agreement then becomes COUNTERSIGNED."
        ),
        tags=["Agreements"],
    )
    def post(self, request, agreement_number):
        with transaction.atomic():
            agreement = get_object_or_404(
                Agreement.objects.select_for_update(),
                agreement_number=agreement_number,
            )

            if (
                agreement.status != Agreement.Status.SIGNED
                or not agreement.student_signed_document
            ):
                return Response(
                    {
                        "error": (
                            "Only agreements with an uploaded "
                            "student-signed copy can be countersigned."
                        )
                    },
                    status=status.HTTP_400_BAD_REQUEST,
                )

            serializer = AgreementSignedUploadSerializer(data=request.data)
            serializer.is_valid(raise_exception=True)
            uploaded = serializer.validated_data["signed_document"]

            agreement.countersigned_document_sha256 = sha256_of(uploaded)
            agreement.countersigned_document.save(
                f"agreement_{agreement.agreement_number}_countersigned.pdf",
                uploaded,
                save=False,
            )
            agreement.countersigned_by = request.user
            agreement.status = Agreement.Status.COUNTERSIGNED
            agreement.board_countersigned_at = timezone.now()
            agreement.save(
                update_fields=[
                    "countersigned_document",
                    "countersigned_document_sha256",
                    "countersigned_by",
                    "status",
                    "board_countersigned_at",
                    "updated_at",
                ]
            )

        return Response(
            {
                "message": "Countersigned agreement uploaded successfully.",
                "agreement": AgreementSerializer(agreement).data,
            },
            status=status.HTTP_200_OK,
        )

# DOWNLOAD AGREEMENT PDF (original, student-signed or countersigned)
# --------------------------------------------------

class AgreementPDFDownloadView(APIView):
    permission_classes = [permissions.IsAuthenticated]

    @extend_schema(
        parameters=[AGREEMENT_NUMBER_PARAM, VERSION_PARAM],
        responses={
            200: OpenApiTypes.BINARY,
            400: ErrorResponseSerializer,
            403: ErrorResponseSerializer,
            404: ErrorResponseSerializer,
        },
        summary="Download an agreement PDF",
        description=(
            "Students can download their own agreements. University "
            "administrators and Loan Board Officers can download all "
            "agreements. Use the 'version' query parameter to choose the "
            "copy: 'original' (default) is what the student signs, "
            "'student_signed' is what the officer countersigns, and "
            "'countersigned' is the final copy."
        ),
        tags=["Agreements"],
    )
    def get(self, request, agreement_number):
        agreement = get_object_or_404(
            Agreement, agreement_number=agreement_number
        )

        role = get_user_role(request.user)

        if role == UserProfile.Role.STUDENT:
            if agreement.student_id not in get_student_identifiers(
                request.user
            ):
                return Response(
                    {
                        "error": (
                            "You are not authorized to download "
                            "this agreement."
                        )
                    },
                    status=status.HTTP_403_FORBIDDEN,
                )
        elif role not in (
            UserProfile.Role.UNIVERSITY_ADMIN,
            UserProfile.Role.LOAN_BOARD_OFFICER,
        ):
            return Response(
                {"error": "You are not authorized to download agreements."},
                status=status.HTTP_403_FORBIDDEN,
            )

        version = request.query_params.get("version", "original")

        if version not in DOWNLOAD_VERSIONS:
            return Response(
                {
                    "error": (
                        "version must be one of: "
                        + ", ".join(DOWNLOAD_VERSIONS)
                        + "."
                    )
                },
                status=status.HTTP_400_BAD_REQUEST,
            )

        field_name, filename_template = DOWNLOAD_VERSIONS[version]
        stored_file = getattr(agreement, field_name)

        if not stored_file:
            return Response(
                {
                    "error": (
                        f"The {version.replace('_', ' ')} copy "
                        "is not available yet."
                    )
                },
                status=status.HTTP_404_NOT_FOUND,
            )

        try:
            pdf_file = stored_file.open("rb")
        except (OSError, ValueError):
            return Response(
                {"error": "The agreement PDF could not be accessed."},
                status=status.HTTP_404_NOT_FOUND,
            )

        return FileResponse(
            pdf_file,
            as_attachment=True,
            filename=filename_template.format(
                number=agreement.agreement_number
            ),
            content_type="application/pdf",
        )

def get_user_role(user):
    # Anonymous requests happen when drf-spectacular builds the Swagger schema,
    # and when unauthenticated clients hit the API. They have no profile.
    if user is None or not getattr(user, "is_authenticated", False):
        return None
    if user.is_superuser:
        return "university_admin"
    try:
        return user.profile.role
    except UserProfile.DoesNotExist:
        return UserProfile.Role.STUDENT

class LoanApplication(models.Model):
    class Status(models.TextChoices):
        PENDING = "pending", "Pending"
        APPROVED = "approved", "Approved"
        REJECTED = "rejected", "Rejected"

    owner = models.ForeignKey(
        settings.AUTH_USER_MODEL,
        on_delete=models.CASCADE,
        related_name="loan_applications",
    )
    loan_reference = models.CharField(
        max_length=30,
        unique=True,
        blank=True,
    )
    tuition_amount = models.DecimalField(
        max_digits=12,
        decimal_places=2,
        default=Decimal("0.00"),
        validators=[MinValueValidator(Decimal("0.00"))],
    )
    upkeep_amount = models.DecimalField(
        max_digits=12,
        decimal_places=2,
        default=Decimal("0.00"),
        validators=[MinValueValidator(Decimal("0.00"))],
    )
    status = models.CharField(
        max_length=10,
        choices=Status.choices,
        default=Status.PENDING,
    )
    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)

    class Meta:
        ordering = ["-created_at"]

    def get_total_amount(self):
        return self.tuition_amount + self.upkeep_amount

    def save(self, *args, **kwargs):
        # First save gets a database ID; then the reference is built from it,
        # e.g. ID 15 in 2026 -> SLBS-2026-00015. Both saves succeed or neither does.
        with transaction.atomic():
            super().save(*args, **kwargs)
            if not self.loan_reference:
                self.loan_reference = f"SLBS-{timezone.now().year}-{self.pk:05d}"
                super().save(update_fields=["loan_reference"])

    def __str__(self):
        return f"Loan application {self.pk} (Total: {self.get_total_amount()})"

from .models import UserProfile, get_user_role


ALL_ROLES = frozenset(UserProfile.Role.values)
LOAN_READER_ROLES = frozenset(
    {
        UserProfile.Role.STUDENT,
        UserProfile.Role.UNIVERSITY_ADMIN,
        UserProfile.Role.LOAN_BOARD_OFFICER,
    }
)


class HasRolePermission(BasePermission):
    action_roles = {
        "metadata": ALL_ROLES,
        "list": LOAN_READER_ROLES,
        "retrieve": LOAN_READER_ROLES,
        "create": frozenset({UserProfile.Role.STUDENT}),
        "update": frozenset(
            {UserProfile.Role.STUDENT, UserProfile.Role.LOAN_BOARD_OFFICER}
        ),
        "partial_update": frozenset(
            {UserProfile.Role.STUDENT, UserProfile.Role.LOAN_BOARD_OFFICER}
        ),
        "destroy": frozenset({UserProfile.Role.STUDENT}),
    }

    def has_permission(self, request, view):
        role = get_user_role(request.user)
        action = getattr(view, "action", None)
        allowed_roles = getattr(view, "allowed_roles", None)
        if allowed_roles is not None:
            return role in allowed_roles
        return role in self.action_roles.get(action, frozenset())

class RegistrationSerializer(serializers.ModelSerializer):
    email = serializers.EmailField(required=True)
    password = serializers.CharField(write_only=True, trim_whitespace=False)
    password_confirmation = serializers.CharField(write_only=True, trim_whitespace=False)
    # Stored on UserProfile, not User. Input-only, so it is never serialized back.
    student_registration_number = serializers.CharField(
        max_length=50,
        required=True,
        write_only=True,
    )

    class Meta:
        model = User
        fields = [
            "id",
            "username",
            "email",
            "student_registration_number",
            "password",
            "password_confirmation",
        ]
        read_only_fields = ["id"]

    def validate_email(self, value):
        value = value.strip().lower()
        if User.objects.filter(email__iexact=value).exists():
            raise serializers.ValidationError("A user with this email already exists.")
        return value

    def validate_student_registration_number(self, value):
        value = value.strip().upper()
        if not value:
            raise serializers.ValidationError("Student registration number is required.")
        if UserProfile.objects.filter(student_registration_number__iexact=value).exists():
            raise serializers.ValidationError(
                "This student registration number is already registered."
            )
        return value

    def validate(self, attrs):
        if attrs["password"] != attrs["password_confirmation"]:
            raise serializers.ValidationError(
                {"password_confirmation": "Passwords do not match."}
            )

        candidate = User(username=attrs.get("username", ""), email=attrs.get("email", ""))
        try:
            password_validation.validate_password(attrs["password"], user=candidate)
        except ValidationError as error:
            raise serializers.ValidationError({"password": list(error.messages)}) from error

        attrs.pop("password_confirmation")
        return attrs

    def create(self, validated_data):
        password = validated_data.pop("password")
        registration_number = validated_data.pop("student_registration_number")
        try:
            with transaction.atomic():  # user + profile succeed or fail together
                user = User.objects.create_user(password=password, **validated_data)
                UserProfile.objects.update_or_create(
                    user=user,
                    defaults={
                        "role": UserProfile.Role.STUDENT,
                        "student_registration_number": registration_number,
                    },
                )
        except IntegrityError:
            raise serializers.ValidationError(
                "Username, email or registration number is already in use."
            )
        return user

class PasswordResetConfirmSerializer(serializers.Serializer):
    email = serializers.EmailField()
    otp = serializers.CharField(max_length=6)
    new_password = serializers.CharField(write_only=True, trim_whitespace=False)
    password_confirmation = serializers.CharField(write_only=True, trim_whitespace=False)

    def validate(self, attrs):
        if attrs["new_password"] != attrs["password_confirmation"]:
            raise serializers.ValidationError(
                {"password_confirmation": "Passwords do not match."}
            )

        # One generic error for every failure, so attackers learn nothing.
        generic = serializers.ValidationError({"otp": "Invalid or expired OTP."})
        email = attrs["email"].strip()

        user = (
            User.objects.filter(email__iexact=email, is_active=True)
            .order_by("id")
            .first()
        )
        record = (
            PasswordResetOTP.objects
            .filter(email__iexact=email, created_at__gte=timezone.now() - OTP_TTL)
            .order_by("-created_at")
            .first()
        )
        if not user or not record or record.attempts >= MAX_OTP_ATTEMPTS:
            raise generic

        # Encode to bytes: compare_digest raises TypeError on non-ASCII str input.
        if not secrets.compare_digest(record.otp.encode(), attrs["otp"].encode()):
            PasswordResetOTP.objects.filter(pk=record.pk).update(attempts=F("attempts") + 1)
            raise generic

        try:
            password_validation.validate_password(attrs["new_password"], user=user)
        except ValidationError as error:
            raise serializers.ValidationError(
                {"new_password": list(error.messages)}
            ) from error

        attrs["user"] = user
        attrs["otp_record"] = record
        return attrs


class RoleTokenObtainPairSerializer(TokenObtainPairSerializer):
    @classmethod
    def get_token(cls, user):
        token = super().get_token(user)
        token["role"] = get_user_role(user)
        return token

    def validate(self, attrs):
        data = super().validate(attrs)
        data["role"] = get_user_role(self.user)
        return data

@extend_schema(
    tags=["Authentication"],
    summary="Request Password Reset",
    description="Sends a one-time password (OTP) to the user's registered email.",
)
class PasswordResetRequestView(APIView):
    permission_classes = [AllowAny]
    authentication_classes = []
    throttle_classes = [PasswordResetRequestThrottle]

    @extend_schema(request=PasswordResetRequestSerializer, responses={200: DetailSerializer})
    def post(self, request):
        serializer = PasswordResetRequestSerializer(data=request.data)
        serializer.is_valid(raise_exception=True)

        user = (
            User.objects.filter(
                email__iexact=serializer.validated_data["email"].strip(),
                is_active=True,
            )
            .order_by("id")
            .first()
        )

        if user:
            otp = f"{secrets.randbelow(1_000_000):06d}"  # keeps leading zeros
            with transaction.atomic():
                PasswordResetOTP.objects.filter(email__iexact=user.email).delete()
                PasswordResetOTP.objects.create(email=user.email, otp=otp)
            try:
                send_mail(
                    subject="Password Reset OTP",
                    message=f"Your OTP is: {otp}. It expires in 10 minutes.",
                    from_email=settings.DEFAULT_FROM_EMAIL,
                    recipient_list=[user.email],
                )
            except Exception:
                # Never reveal mail failures: the response must look identical.
                logger.exception("Failed to send password reset email")

        return Response(
            {"detail": "If an account exists for this email, an OTP has been sent."}
        )


class PasswordResetConfirmView(APIView):
    permission_classes = [AllowAny]
    authentication_classes = []
    throttle_classes = [PasswordResetConfirmThrottle]

    @extend_schema(
        tags=["Authentication"],
        summary="Confirm Password Reset",
        description="Verifies OTP and resets the password.",
        request=PasswordResetConfirmSerializer,
        responses={200: DetailSerializer},
    )
    def post(self, request):
        serializer = PasswordResetConfirmSerializer(data=request.data)
        serializer.is_valid(raise_exception=True)

        user = serializer.validated_data["user"]

        with transaction.atomic():
            user.set_password(serializer.validated_data["new_password"])
            user.save(update_fields=["password"])
            # Single use: remove every OTP for this email.
            PasswordResetOTP.objects.filter(email__iexact=user.email).delete()

        return Response({"detail": "Password has been reset successfully."})

@extend_schema(tags=["Loan Applications"], description="Manage loan applications.")
class LoanApplicationViewSet(viewsets.ModelViewSet):
    permission_classes = [IsAuthenticated, HasRolePermission]

    def get_queryset(self):
        role = get_user_role(self.request.user)
        if role in {
            UserProfile.Role.UNIVERSITY_ADMIN,
            UserProfile.Role.LOAN_BOARD_OFFICER,
        }:
            return LoanApplication.objects.all()
        if role == UserProfile.Role.STUDENT:
            return LoanApplication.objects.filter(owner=self.request.user)
        return LoanApplication.objects.none()

    def get_serializer_class(self):
        if (
            self.action in {"update", "partial_update"}
            and get_user_role(self.request.user) == UserProfile.Role.LOAN_BOARD_OFFICER
        ):
            return LoanApplicationReviewSerializer
        return LoanApplicationSerializer

    def perform_create(self, serializer):
        serializer.save(owner=self.request.user)

    def perform_update(self, serializer):
        if (
            get_user_role(self.request.user) == UserProfile.Role.STUDENT
            and serializer.instance.status != LoanApplication.Status.PENDING
        ):
            raise PermissionDenied("Only pending applications can be edited.")
        serializer.save()

    def perform_destroy(self, instance):
        if instance.status != LoanApplication.Status.PENDING:
            raise PermissionDenied("Only pending applications can be deleted.")
        instance.delete()

class BondStatusUpdateSerializer(serializers.ModelSerializer):
    status = serializers.ChoiceField(
        choices=["approved", "rejected"],
        help_text="The decision for the bond: approved or rejected.",
    )

    class Meta:
        model = Bond

        fields = [
            "status",
            "rejection_reason",
        ]

        extra_kwargs = {
            "rejection_reason": {
                "help_text": (
                    "Required when status is rejected. "
                    "Ignored (cleared) when the bond is approved."
                ),
            },
        }

    def validate(self, attrs):
        if self.instance is None or self.instance.status != "pending":
            raise serializers.ValidationError(
                "Only pending bonds can be reviewed."
            )

        new_status = attrs.get("status")

        if new_status is None:
            raise serializers.ValidationError({
                "status": "This field is required."
            })

        if new_status not in ("approved", "rejected"):
            raise serializers.ValidationError({
                "status": "A bond can only be approved or rejected."
            })

        reason = attrs.get(
            "rejection_reason",
            self.instance.rejection_reason,
        )

        if new_status == "rejected":
            if not reason or not reason.strip():
                raise serializers.ValidationError({
                    "rejection_reason": (
                        "A rejection reason is required."
                    )
                })
            attrs["rejection_reason"] = reason.strip()

        if new_status == "approved":
            attrs["rejection_reason"] = ""

        return attrs

    permission_classes = [permissions.IsAuthenticated, IsUniversityAdmin]
    http_method_names = ["patch", "head", "options"]
    queryset = Bond.objects.all()

    def update(self, request, *args, **kwargs):
        partial = kwargs.pop("partial", True)

        with transaction.atomic():
            bond = get_object_or_404(
                Bond.objects.select_for_update(), pk=kwargs["pk"]
            )

            if bond.status != "pending":
                raise ValidationError(
                    {
                        "status": (
                            f"This bond is already {bond.status} "
                            "and cannot be reviewed again."
                        )
                    }
                )

            serializer = self.get_serializer(
                bond, data=request.data, partial=partial
            )
            serializer.is_valid(raise_exception=True)
            serializer.save(
                reviewed_by=request.user,
                reviewed_at=timezone.now(),
            )

        return Response(serializer.data)

def tag_authentication_endpoints(result, generator, request, public):
    """drf-spectacular post-processing hook."""
    for path, path_item in result.get("paths", {}).items():
        if not path.startswith(AUTH_PATH_PREFIXES):

        for operation in path_item.values():

        tags=[ADMIN_TAG],

        description=(
            "University administrators only. Streams the uploaded file so it can be "

        ),
        parameters=[DOCUMENT_ID_PARAM],
        responses={
            200: OpenApiTypes.BINARY,
            401: UNAUTHENTICATED,
            403: ADMIN_ONLY,
            404: OpenApiResponse(
                description="Document or file not found."
            ),
        },
    )
    def get(self, request, pk):
        document = get_object_or_404(Document, pk=pk)

        if not document.file:
            raise Http404("No file uploaded for this document.")

        try:
            file_handle = document.file.open("rb")
        except (OSError, ValueError):
            raise Http404("The document file could not be accessed.")

        content_type, _ = mimetypes.guess_type(document.file.name)

        return FileResponse(
            file_handle,
            as_attachment=False,
            filename=document.file.name.split("/")[-1],
            content_type=content_type or "application/octet-stream",
        )


class DocumentVerificationAPIView(APIView):
    permission_classes = [permissions.IsAuthenticated, IsUniversityAdmin]

    @extend_schema(
        tags=[ADMIN_TAG],
        summary="Verify or reject a document",
        description=(
            "University administrators only. Marks a document as "
            "'verified' or 'rejected'. A rejection must include a reason. "
            "The student automatically receives a notification with the "
            "decision. When this verification completes the student's set "
            "of compulsory documents, one pending bond is generated "
            "automatically (never a duplicate) and the student is notified "
            "again; the response then includes the new bond's ID."
        ),
        parameters=[DOCUMENT_ID_PARAM],
        request=DocumentVerificationSerializer,
        responses={
            200: DocumentVerificationResultSerializer,
            400: OpenApiResponse(
                description="Invalid status or missing rejection reason."
            ),
            401: UNAUTHENTICATED,
            403: ADMIN_ONLY,
            404: OpenApiResponse(description="Document not found."),
        },
    )
    def patch(self, request, pk):
        document = get_object_or_404(
            Document.objects.select_related("student"), pk=pk
        )

        serializer = DocumentVerificationSerializer(data=request.data)
        serializer.is_valid(raise_exception=True)

        new_status = serializer.validated_data["status"]
        reason = serializer.validated_data["rejection_reason"]
        label = document.get_document_type_display()

        if new_status == "verified":
            message = f"Your {label} has been verified."
        else:
            message = f"Your {label} was rejected. Reason: {reason}"

        bond = None

        with transaction.atomic():
            document.status = new_status
            document.rejection_reason = reason
            document.reviewed_by = request.user
            document.reviewed_at = timezone.now()

            document.save(
                update_fields=[
                    "status",
                    "rejection_reason",
                    "reviewed_by",
                    "reviewed_at",
                ]
            )

            Notification.objects.create(
                student=document.student,
                document=document,
                message=message,
            )

            if new_status == "verified":
                bond = generate_bond_if_eligible(document.student, document)

        result = DocumentVerificationResultSerializer(
            {
                "id": document.id,
                "status": document.status,
                "rejection_reason": document.rejection_reason,
                "reviewed_by": request.user.username,
                "reviewed_at": document.reviewed_at,
                "bond_generated": bond is not None,
                "bond_id": bond.id if bond else None,
                "bond_reference": bond.bond_reference if bond else None,
            }
        )

        return Response(result.data)

from django.db import IntegrityError, transaction

from bonds.models import Bond

from .models import Document, Notification

# Document types a student must have verified before a bond is generated.
# All other types (affidavit, guarantor_form, guarantor_id, other_letter)
# are optional. Edit this tuple to change the rule.
COMPULSORY_DOCUMENT_TYPES = (
    "student_id",
    "national_id",
)

# A student with a bond in any of these states never gets another one
# automatically. A rejected bond does not block a new one.
BLOCKING_BOND_STATUSES = ("pending", "approved", "completed")


def has_all_compulsory_documents_verified(student):
    verified_types = set(
        Document.objects.filter(
            student=student,
            status="verified",
            document_type__in=COMPULSORY_DOCUMENT_TYPES,
        ).values_list("document_type", flat=True)
    )

    return verified_types.issuperset(COMPULSORY_DOCUMENT_TYPES)


def generate_bond_if_eligible(student, triggering_document):
    """
    Create one pending bond when the student's compulsory documents are all
    verified. Returns the new Bond, or None if nothing was created.

    Call this inside the same transaction as the document review so the
    status change, notifications and bond are saved together.
    """
    if triggering_document.document_type not in COMPULSORY_DOCUMENT_TYPES:
        return None

    if not has_all_compulsory_documents_verified(student):
        return None

    if Bond.objects.filter(
        student=student, status__in=BLOCKING_BOND_STATUSES
    ).exists():
        return None

    try:
        # Savepoint, so a race lost against the database's unique
        # constraint does not break the surrounding transaction.
        with transaction.atomic():
            bond = Bond.objects.create(student=student, status="pending")
    except IntegrityError:
        return None

    Notification.objects.create(
        student=student,
        document=triggering_document,
        message=(
            "All your compulsory documents have been verified. "
            f"Bond {bond.bond_reference} has been generated and is "
            "awaiting approval by a university administrator."
        ),
    )

    return bond

import shutil

User = get_user_model()

TEST_MEDIA_ROOT = tempfile.mkdtemp()


def make_user(username, role, registration_number=None):
    """
    ASSUMPTION: UserProfile has a `user` link, a `role` field and a
    `student_registration_number` field. If your model differs, this is the
    only place to adjust.
    """

    profile, _ = UserProfile.objects.get_or_create(user=user)
    profile.role = role
    if registration_number:
        profile.student_registration_number = registration_number
    profile.save()

@override_settings(MEDIA_ROOT=TEST_MEDIA_ROOT)
class WorkflowIntegrationTests(APITestCase):

    @classmethod
    def tearDownClass(cls):
        super().tearDownClass()
        shutil.rmtree(TEST_MEDIA_ROOT, ignore_errors=True)

    def setUp(self):
        self.student = make_user(
            "student1", UserProfile.Role.STUDENT, "REG-001"
        )
        self.other_student = make_user(
            "student2", UserProfile.Role.STUDENT, "REG-002"
        )
        self.admin = make_user("admin1", UserProfile.Role.UNIVERSITY_ADMIN)
        self.officer = make_user(
            "officer1", UserProfile.Role.LOAN_BOARD_OFFICER
        )

    # ---------- helpers ----------

    def upload(self, doc_type, user=None):
        self.client.force_authenticate(user or self.student)
        response = self.client.post(
            reverse("student_documents_api"),
            {
                "document_type": doc_type,
                "file": SimpleUploadedFile(
                    f"{doc_type}.pdf",
                    b"%PDF-1.4 test file",
                    content_type="application/pdf",
                ),
            },
            format="multipart",
        )
        self.assertEqual(response.status_code, 201, response.data)
        return response.data["id"]

    def review(self, doc_id, new_status="verified", reason=""):
        self.client.force_authenticate(self.admin)
        return self.client.patch(
            reverse("document_verification_api", kwargs={"pk": doc_id}),
            {"status": new_status, "rejection_reason": reason},
            format="json",
        )

    def upload_and_verify_all(self, user=None):
        """Returns the response of the LAST verification."""
        response = None
        for doc_type in COMPULSORY_DOCUMENT_TYPES:
            doc_id = self.upload(doc_type, user)
            response = self.review(doc_id)
            self.assertEqual(response.status_code, 200, response.data)
        return response

    def approve_bond(self, bond_id):
        self.client.force_authenticate(self.admin)
        return self.client.patch(
            reverse("bond-status-update", kwargs={"pk": bond_id}),
            {"status": "approved"},
            format="json",
        )

    # ---------- tests ----------

    def test_full_workflow(self):
        # 1-2. Student submits, admin verifies. The bond appears only
        # after the last compulsory document is verified.
        last = None
        for index, doc_type in enumerate(COMPULSORY_DOCUMENT_TYPES):
            last = self.review(self.upload(doc_type))
            self.assertEqual(last.status_code, 200, last.data)
            is_last = index == len(COMPULSORY_DOCUMENT_TYPES) - 1
            self.assertEqual(last.data["bond_generated"], is_last)
            self.assertEqual(Bond.objects.count(), 1 if is_last else 0)

        # 3. Exactly one pending bond, reported in the response.
        bond = Bond.objects.get()
        self.assertEqual(bond.student, self.student)
        self.assertEqual(bond.status, "pending")
        self.assertEqual(last.data["bond_id"], bond.id)

        # Notifications: one per review plus one for the bond.
        self.assertEqual(
            Notification.objects.filter(student=self.student).count(),
            len(COMPULSORY_DOCUMENT_TYPES) + 1,
        )

        # The student sees the bond; another student does not.
        self.client.force_authenticate(self.student)
        listing = self.client.get(reverse("bond-list-create"))
        self.assertEqual([b["id"] for b in listing.data], [bond.id])
        self.client.force_authenticate(self.other_student)
        self.assertEqual(
            self.client.get(reverse("bond-list-create")).data, []
        )

        # 5. No agreement before the bond is approved.
        self.client.force_authenticate(self.student)
        response = self.client.post(reverse("agreement-generate"))
        self.assertEqual(response.status_code, 400)

        # 4. Admin approves the bond.
        response = self.approve_bond(bond.id)
        self.assertEqual(response.status_code, 200, response.data)

        # 5. Student generates the agreement (once only).
        self.client.force_authenticate(self.student)
        response = self.client.post(reverse("agreement-generate"))
        self.assertEqual(response.status_code, 201, response.data)
        agreement_number = response.data["agreement_number"]
        self.assertEqual(Agreement.objects.get().bond_id, bond.id)
        self.assertEqual(
            self.client.post(reverse("agreement-generate")).status_code, 409
        )

        sign_url = reverse(
            "agreement-sign", kwargs={"agreement_number": agreement_number}
        )
        countersign_url = reverse(
            "agreement-countersign",
            kwargs={"agreement_number": agreement_number},
        )

        # Countersigning before the student signs is refused.
        self.client.force_authenticate(self.officer)
        self.assertEqual(self.client.post(countersign_url).status_code, 400)

        # Another student cannot sign it.
        self.client.force_authenticate(self.other_student)
        self.assertEqual(self.client.post(sign_url).status_code, 403)

        # 6. The owner signs.
        self.client.force_authenticate(self.student)
        self.assertEqual(self.client.post(sign_url).status_code, 200)

        # Only the Loan Board Officer may countersign.
        for user in (self.student, self.admin):
            self.client.force_authenticate(user)
            self.assertEqual(
                self.client.post(countersign_url).status_code, 403
            )

        # 7. The officer countersigns.
        self.client.force_authenticate(self.officer)
        self.assertEqual(self.client.post(countersign_url).status_code, 200)
        self.assertEqual(
            Agreement.objects.get().status, Agreement.Status.COUNTERSIGNED
        )

        # The PDF can be downloaded.
        self.client.force_authenticate(self.student)
        response = self.client.get(
            reverse(
                "agreement-download",
                kwargs={"agreement_number": agreement_number},
            )
        )
        self.assertEqual(response.status_code, 200)

    def test_rejected_document_blocks_the_bond(self):
        for doc_type in COMPULSORY_DOCUMENT_TYPES[:-1]:
            self.review(self.upload(doc_type))

        rejected = self.review(
            self.upload(COMPULSORY_DOCUMENT_TYPES[-1]),
            "rejected",
            "Photo is blurry",
        )
        self.assertEqual(rejected.status_code, 200)
        self.assertEqual(Bond.objects.count(), 0)

        # Re-uploading and verifying a good copy completes the set.
        response = self.review(self.upload(COMPULSORY_DOCUMENT_TYPES[-1]))
        self.assertTrue(response.data["bond_generated"])
        self.assertEqual(Bond.objects.count(), 1)

    def test_no_duplicate_bond(self):
        self.upload_and_verify_all()
        self.assertEqual(Bond.objects.count(), 1)

        # Re-verifying more copies of compulsory documents adds nothing.
        for doc_type in COMPULSORY_DOCUMENT_TYPES:
            response = self.review(self.upload(doc_type))
            self.assertFalse(response.data["bond_generated"])

        self.assertEqual(Bond.objects.filter(student=self.student).count(), 1)

    def test_bonds_are_per_student(self):
        self.upload_and_verify_all(self.student)
        self.upload_and_verify_all(self.other_student)
        self.assertEqual(Bond.objects.count(), 2)

    def test_roles_are_enforced_on_documents(self):
        doc_id = self.upload(COMPULSORY_DOCUMENT_TYPES[0])
        verify_url = reverse("document_verification_api", kwargs={"pk": doc_id})

        for user in (self.student, self.officer):
            self.client.force_authenticate(user)
            response = self.client.patch(
                verify_url, {"status": "verified"}, format="json"
            )
            self.assertEqual(response.status_code, 403)

        for user in (self.admin, self.officer):
            self.client.force_authenticate(user)
            response = self.client.get(reverse("student_documents_api"))
            self.assertEqual(response.status_code, 403)

        self.client.force_authenticate(None)
        response = self.client.get(reverse("student_documents_api"))
        self.assertEqual(response.status_code, 401)

    def test_bond_review_is_admin_only(self):
        self.upload_and_verify_all()
        bond = Bond.objects.get()

        for user in (self.student, self.officer):
            self.client.force_authenticate(user)

                reverse("bond-status-update", kwargs={"pk": bond.id}),
                {"status": "approved"},
                format="json",
            )
            self.assertEqual(response.status_code, 403)

import os

from django.core.exceptions import ValidationError

ALLOWED_EXTENSIONS = (".pdf", ".jpg", ".jpeg", ".png")
MAX_UPLOAD_SIZE = 5 * 1024 * 1024  # 5 MB


def validate_document_file(uploaded_file):
    """Accept only PDF/JPG/PNG files up to 5 MB."""
    extension = os.path.splitext(uploaded_file.name)[1].lower()

    if extension not in ALLOWED_EXTENSIONS:
        raise ValidationError(
            "Unsupported file type. Upload a PDF, JPG or PNG file."
        )

    if uploaded_file.size > MAX_UPLOAD_SIZE:
        raise ValidationError(
            "File is too large. The maximum size is 5 MB."
        )

from django.contrib import messages
from django.contrib.auth import views as auth_views
from django.contrib.auth.decorators import login_required
from django.core.exceptions import PermissionDenied
from django.core.exceptions import ValidationError as DjangoValidationError
from django.shortcuts import redirect, render
from django.urls import reverse

from .models import Document
from .validators import validate_document_file


class RoleBasedLoginView(auth_views.LoginView):
    template_name = "registration/login.html"

    def get_success_url(self):
        if self.request.user.is_staff:
            return "/admin/"

        return reverse("submit_document")


@login_required
def submit_document(request):
    if request.user.is_staff:
        raise PermissionDenied(
            "Administrators cannot submit student documents."
        )

    if request.method == "POST":
        document_type = request.POST.get("document_type")
        uploaded_file = request.FILES.get("file")

        valid_types = {value for value, _ in Document.DOCUMENT_TYPES}

        if document_type not in valid_types or not uploaded_file:
            messages.error(
                request,
                "Please select a document type and upload a file.",
            )
            return redirect("submit_document")

        try:
            validate_document_file(uploaded_file)
        except DjangoValidationError as error:
            messages.error(request, error.messages[0])
            return redirect("submit_document")

        Document.objects.create(
            student=request.user,
            document_type=document_type,
            file=uploaded_file,
            status="pending",
        )

        messages.success(request, "Document submitted successfully.")

        return redirect("submit_document")

    return render(request, "documents/submit_document.html")

from rest_framework.views import APIView
from rest_framework.permissions import IsAuthenticated
from rest_framework.response import Response
from rest_framework import status


class ReportingPermissionMixin:
    permission_classes = [IsAuthenticated]

    def has_reporting_access(self, request):
        return request.user.groups.filter(
            name="loan_board_officer"
        ).exists()

    def unavailable_response(self, report_name):
        return Response(
            {
                "report": report_name,
                "message": "Report data integration is pending."
            },
            status=status.HTTP_503_SERVICE_UNAVAILABLE
        )


class ActiveLoansReportView(ReportingPermissionMixin, APIView):

    def get(self, request):
        if not self.has_reporting_access(request):
            return Response(
                {"error": "Reporting access denied."},
                status=status.HTTP_403_FORBIDDEN
            )

        return self.unavailable_response("active_loans")

from django.test import TestCase


from django.contrib.auth import get_user_model
from django.contrib.auth.models import Group
from django.test import TestCase
from rest_framework.test import APIClient


class BondVerificationTests(TestCase):

    def setUp(self):
        self.client = APIClient()
        self.url = "/api/verification/bonds/12/"
        self.user_model = get_user_model()

    def create_user(self, username, role=None):
        user = self.user_model.objects.create_user(
            username=username,
            password="TestPassword123!"
        )

        if role:
            group, created = Group.objects.get_or_create(
                name=role
            )
            user.groups.add(group)

        return user

    def test_unauthenticated_user(self):
        response = self.client.get(
            self.url,
            HTTP_HOST="localhost"
        )

        self.assertIn(response.status_code, [401, 403])

    def test_student_cannot_verify(self):
        student = self.create_user("student")

        self.client.force_authenticate(user=student)

        response = self.client.get(
            self.url,
            HTTP_HOST="localhost"
        )

        self.assertEqual(response.status_code, 403)

    def test_university_admin_access(self):
        admin = self.create_user(
            "university_admin_test",
            "university_admin"
        )

        self.client.force_authenticate(user=admin)

        response = self.client.get(
            self.url,
            HTTP_HOST="localhost"
        )

        self.assertEqual(response.status_code, 503)

    def test_employer_access(self):
        employer = self.create_user(
            "employer_test",
            "employer"
        )

        self.client.force_authenticate(user=employer)

        response = self.client.get(
            self.url,
            HTTP_HOST="localhost"
        )

        self.assertEqual(response.status_code, 503)

from rest_framework.views import APIView
from rest_framework.permissions import IsAuthenticated
from rest_framework.response import Response
from rest_framework import status


class StudentBondVerificationView(APIView):
    permission_classes = [IsAuthenticated]

    def get(self, request, student_id):
        # Only university admins and employers can verify bonds.
        allowed_roles = ["university_admin", "employer"]

        is_allowed = request.user.groups.filter(
            name__in=allowed_roles
        ).exists()

        if not is_allowed:
            return Response(
                {"error": "You are not authorized to verify bonds."},
                status=status.HTTP_403_FORBIDDEN
            )

        # Bond records will be connected later.
        return Response(
            {
                "message": "Bond verification is not yet connected.",
                "student_id": student_id
            },
            status=status.HTTP_503_SERVICE_UNAVAILABLE
        )
